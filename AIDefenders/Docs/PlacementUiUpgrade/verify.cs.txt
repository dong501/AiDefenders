var report=new System.Collections.Generic.List<string>();
foreach(string path in new[]{"Assets/Scenes/GameStages/City/City_1.unity","Assets/Scenes/GameStages/Desert/Desert_1.unity"})
{
 var s=UnityEditor.SceneManagement.EditorSceneManager.OpenScene(path,UnityEditor.SceneManagement.OpenSceneMode.Additive);
 try
 {
  var all=s.GetRootGameObjects().SelectMany(g=>g.GetComponentsInChildren<Transform>(true)).ToArray();
  Physics.SyncTransforms();
  if(path.Contains("City"))
  {
   var surfaces=all.Select(t=>t.GetComponent<TurretBuildSurface>()).Where(t=>t && t.gameObject.activeInHierarchy).ToArray();int roofs=0,walks=0;
   foreach(var surface in surfaces)
   {
    var c=surface.GetComponent<Collider>();var b=c.bounds;RaycastHit hit;
    if(c.Raycast(new Ray(new Vector3(b.center.x,b.max.y+2,b.center.z),Vector3.down),out hit,b.size.y+4) && TurretPlacementRules.Allows(hit,Vector3.one*.5f))
    {if(c is MeshCollider)roofs++;else walks++;}
   }
   if(roofs<20 || walks<80)throw new Exception("Insufficient build surfaces: "+roofs+"/"+walks);
   report.Add("City: buildable roof centers="+roofs+", sidewalks="+walks);
   var paths=all.Select(t=>t.GetComponent<EnemyPath>()).Where(t=>t && t.gameObject.activeInHierarchy).ToArray();int blocked=0;
   foreach(var p in paths){var points=p.GetWorldPoints();for(int i=1;i<points.Length;i++)for(int j=0;j<=10;j++){var q=Vector3.Lerp(points[i-1],points[i],j/10f);foreach(var surface in surfaces.Where(t=>t.name=="Buildable_Sidewalk")){var b=surface.GetComponent<Collider>().bounds;if(q.x>b.min.x-1 && q.x<b.max.x+1 && q.z>b.min.z-1 && q.z<b.max.z+1)blocked++;}}}
   report.Add("Sidewalks intersecting enemy center lanes="+blocked);if(blocked>0)throw new Exception("Sidewalk route intersection");
  }
  else
  {
   var terrain=all.Select(t=>t.GetComponent<Terrain>()).First(t=>t);var c=terrain.GetComponent<TerrainCollider>();int accepted=0,rejected=0;
   foreach(var marker in all.Where(t=>t.name.StartsWith("Natural_Firing_Terrace_")))
   {RaycastHit hit;if(c.Raycast(new Ray(marker.position+Vector3.up*5,Vector3.down),out hit,12) && TurretPlacementRules.Allows(hit,Vector3.one*.5f))accepted++;}
   foreach(var p in all.Select(t=>t.GetComponent<EnemyPath>()).Where(t=>t && t.gameObject.activeInHierarchy))foreach(var point in p.GetWorldPoints())
   {RaycastHit hit;if(c.Raycast(new Ray(point+Vector3.up*4,Vector3.down),out hit,8)){if(TurretPlacementRules.Allows(hit,Vector3.one*.5f))throw new Exception("Canyon floor accepted");rejected++;}}
   if(accepted<6)throw new Exception("Terraces rejected");report.Add("Desert: high ground accepted="+accepted+", canyon lane points rejected="+rejected);
  }
 }
 finally{UnityEditor.SceneManagement.EditorSceneManager.CloseScene(s,true);}
}
var keyMethod=typeof(TurretPaint).GetMethod("Key",System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.NonPublic);
var prefabs=UnityEditor.AssetDatabase.FindAssets("t:Prefab",new[]{"Assets/Turrets"}).Select(id=>UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>(UnityEditor.AssetDatabase.GUIDToAssetPath(id))).Where(p=>p.GetComponent<TurretController>() && p.GetComponent<TurretController>().upgradeTiers!=null && p.GetComponent<TurretController>().upgradeTiers.Length==3).ToArray();
int painted=0;
foreach(var prefab in prefabs)
{
 var instance=UnityEngine.Object.Instantiate(prefab);var turret=instance.GetComponent<TurretController>();turret.sourcePrefab=prefab;
 string key=(string)keyMethod.Invoke(null,new object[]{turret});bool existed=PlayerPrefs.HasKey(key);int previous=PlayerPrefs.GetInt(key);
 try
 {
  PlayerPrefs.SetInt(key,3);TurretPaint.Apply(turret);bool changed=false;
  foreach(var renderer in instance.GetComponentsInChildren<MeshRenderer>(true))for(int i=0;i<renderer.sharedMaterials.Length;i++)
  {var block=new MaterialPropertyBlock();renderer.GetPropertyBlock(block,i);var color=block.GetColor("_BaseColor");if(color.r>color.b*1.5f && color.a>.1f)changed=true;}
  if(!changed)throw new Exception("No painted armor: "+prefab.name);
  PlayerPrefs.SetInt(key,0);TurretPaint.Apply(turret);painted++;
 }
 finally{if(existed)PlayerPrefs.SetInt(key,previous);else PlayerPrefs.DeleteKey(key);UnityEngine.Object.DestroyImmediate(instance);}
}
report.Add("Paint applied and default restored on "+painted+" turret prefabs; original preferences restored");
System.IO.File.WriteAllLines("Docs/PlacementUiUpgrade/Verification.txt",report);
return string.Join("\n",report.ToArray());
