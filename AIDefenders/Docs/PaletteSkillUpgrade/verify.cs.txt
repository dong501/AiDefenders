var report=new System.Collections.Generic.List<string>();
if(TurretPaint.Colors.Length!=65 || TurretPaint.Colors.Skip(1).Select(c=>ColorUtility.ToHtmlStringRGB(c)).Distinct().Count()!=64)throw new Exception("Palette must have 64 distinct colors plus reset");
var keyMethod=typeof(TurretPaint).GetMethod("Key",System.Reflection.BindingFlags.Static|System.Reflection.BindingFlags.NonPublic);
int models=0,paintedSlots=0,protectedSlots=0;
foreach(string file in TurretTierModelConnection.Files.Concat(new[]{"Turret_Test MG"}))
{
 var prefab=UnityEditor.AssetDatabase.LoadAssetAtPath<GameObject>("Assets/Turrets/Turret_Test/"+file+".prefab");
 var instance=UnityEngine.Object.Instantiate(prefab);var turret=instance.GetComponent<TurretController>();turret.sourcePrefab=prefab;
 string key=(string)keyMethod.Invoke(null,new object[]{turret})+"_Palette64";bool exists=PlayerPrefs.HasKey(key);int previous=PlayerPrefs.GetInt(key);
 try
 {
  for(int choice=0;choice<=64;choice++)
  {
   PlayerPrefs.SetInt(key,choice);TurretPaint.Apply(turret);
   foreach(var renderer in instance.GetComponentsInChildren<MeshRenderer>(true))
   {
    var slots=renderer.GetComponent<TurretPaintSlots>();if(!slots)throw new Exception("Missing paint mask");
    for(int slot=0;slot<renderer.sharedMaterials.Length;slot++)
    {
     Color original=renderer.sharedMaterials[slot].GetColor("_BaseColor");var block=new MaterialPropertyBlock();renderer.GetPropertyBlock(block,slot);Color actual=block.GetColor("_BaseColor");
     bool painted=System.Array.IndexOf(slots.materialSlots,slot)>=0;
     if(!painted || choice==0){if(((Vector4)(actual-original)).sqrMagnitude>.000001f)throw new Exception("Protected color changed: "+file+"/"+renderer.name+"/"+slot);}
     else{Color expected=TurretPaint.Colors[choice]*Mathf.Lerp(.65f,1f,original.maxColorComponent);expected.a=original.a;if(((Vector4)(actual-expected)).sqrMagnitude>.000001f)throw new Exception("Palette color mismatch");}
     if(choice==1){if(painted)paintedSlots++;else protectedSlots++;}
    }
   }
  }
  models+=turret.upgradeTiers.Length;
 }
 finally{if(exists)PlayerPrefs.SetInt(key,previous);else PlayerPrefs.DeleteKey(key);UnityEngine.Object.DestroyImmediate(instance);}
}
report.Add("64 unique colors + original reset. Tested all 65 choices on "+models+" tier models (includes legacy machine gun).");
report.Add("Exterior slots="+paintedSlots+"; protected weapon/trim/emissive slots="+protectedSlots+". No protected color changed. Preferences restored.");
var scene=UnityEditor.SceneManagement.EditorSceneManager.OpenScene("Assets/Scenes/GameStages/City/City_1.unity",UnityEditor.SceneManagement.OpenSceneMode.Additive);
try{var root=scene.GetRootGameObjects().First(g=>g.name=="City_Green_Guide_Signs");var texts=root.GetComponentsInChildren<TMPro.TMP_Text>();if(root.transform.childCount!=6 || texts.Any(t=>t.text.Contains("중앙대로") || t.text.Contains("시민대로")))throw new Exception("Old signs remain");report.Add("Six Yeongdeungpo street signs: "+string.Join("; ",texts.Where(t=>t.name=="Road_Name").Select(t=>t.text).ToArray()));}
finally{UnityEditor.SceneManagement.EditorSceneManager.CloseScene(scene,true);}
System.IO.File.WriteAllLines("Docs/PaletteSkillUpgrade/Verification.txt",report);return string.Join("\n",report.ToArray());
